   116→function EquityChart({ months }: { months: Record<string, number> }) {
   117→  const pts = useMemo(() => {
   118→    const keys = Object.keys(months).sort()
   119→    return keys.reduce<{ k: string; v: number }[]>((acc, k) => {
   120→      const prev = acc.length > 0 ? acc[acc.length - 1].v : 0
   121→      acc.push({ k, v: prev + months[k] })
   122→      return acc
   123→    }, [])
   124→  }, [months])
   125→  if (pts.length < 2) return null
   126→  const W = 860, H = 200, PAD = 8
   127→  const vals = pts.map((p) => p.v)
   128→  const min = Math.min(...vals, 0), max = Math.max(...vals)
   129→  const x = (i: number) => PAD + (i / (pts.length - 1)) * (W - 2 * PAD)
   130→  const y = (v: number) => H - PAD - ((v - min) / (max - min || 1)) * (H - 2 * PAD)
   131→  const line = pts.map((p, i) => `${x(i).toFixed(1)},${y(p.v).toFixed(1)}`).join(' ')
   132→  const area = `${PAD},${H - PAD} ${line} ${W - PAD},${H - PAD}`
   133→  const yearMarks = [2023, 2024, 2025, 2026].map((yr) => {
   134→    const i = pts.findIndex((p) => p.k.startsWith(String(yr)))
   135→    return i >= 0 ? { i, yr } : null
   136→  }).filter(Boolean)
   137→  return (
   138→    <div className="w-full">
   139→      <svg viewBox={`0 0 ${W} ${H}`} className="h-44 w-full sm:h-52" role="img" aria-label="月度累计盈亏曲线" preserveAspectRatio="none">
   140→        <polygon points={area} fill="rgb(16 185 129 / 0.12)" />
   141→        <line x1={PAD} x2={W - PAD} y1={y(0)} y2={y(0)} stroke="rgb(113 113 122 / 0.4)" strokeDasharray="3 4" strokeWidth="1" />
   142→        <polyline points={line} fill="none" stroke="rgb(52 211 153)" strokeWidth="2" strokeLinejoin="round" />
   143→        {yearMarks.map((m) => (
   144→          <line key={m!.yr} x1={x(m!.i)} x2={x(m!.i)} y1={PAD} y2={H - PAD} stroke="rgb(113 113 122 / 0.35)" strokeWidth="1" strokeDasharray="2 4" />
   145→        ))}
   146→      </svg>
   147→      <div className="mt-1 flex justify-between text-[10px] text-zinc-600">
   148→        {yearMarks.map((m) => <span key={m!.yr}>{m!.yr}</span>)}
   149→        <span>终值 {fmtUSD(vals[vals.length - 1])}</span>
   150→      </div>
   151→    </div>
   152→  )
   153→}
